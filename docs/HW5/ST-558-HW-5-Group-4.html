  1. Next, we’ll create a function that allows a user to input a topic, a date, and an API key.
search_it <- function(topic,date,APIkey){
  if(!is.character(topic)|!is.character(date)|!is.character(APIkey)){
    stop("At least one of these is NOT a string. Use quotation marks and '%20' instead of a space for better results. Also, make sure the date is of the format yyyy-mm-dd.")
  }
  lengthy_url = GET(paste("https://newsapi.org/v2/everything?q=",
                          topic,"&from=",date,"&language=en&pageSize=50&apiKey=",APIkey,sep=''))
  parsed_search = fromJSON(rawToChar(lengthy_url$content))
  top_articles = as.tibble(parsed_search$articles)
  return(top_articles[-1])
}
